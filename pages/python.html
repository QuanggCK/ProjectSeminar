Cú Pháp Cơ Bản

Biến & Kiểu dữ liệu (Dynamic Typing) ▼

Python sử dụng dynamic typing — bạn không cần khai báo kiểu, Python tự nhận diện.

Python 
# Python tự nhận kiểu dữ liệu
ten = "Nguyen Van A"      # str (chuỗi)
tuoi = 20                  # int (số nguyên)
diem_tb = 8.5              # float (số thực)
la_sv = True               # bool (đúng/sai)
danh_sach = [1, 2, 3]      # list (danh sách)
thong_tin = {"a": 1}       # dict (từ điển)

# Kiểm tra kiểu
print(f"ten: {ten} - {type(ten)}")
print(f"tuoi: {tuoi} - {type(tuoi)}")
print(f"diem_tb: {diem_tb} - {type(diem_tb)}")
print(f"la_sv: {la_sv} - {type(la_sv)}")

# String formatting (f-string)
print(f"\nXin chao {ten}, ban {tuoi} tuoi, diem {diem_tb}")
ten: Nguyen Van A - <class 'str'> tuoi: 20 - <class 'int'> diem_tb: 8.5 - <class 'float'> la_sv: True - <class 'bool'> Xin chao Nguyen Van A, ban 20 tuoi, diem 8.5

Hàm (Functions)

Python 
# === Hàm cơ bản ===
def chao(ten):
    """Ham chao - co docstring"""
    return f"Xin chao, {ten}!"

print(chao("An"))

# === Tham số mặc định ===
def tinh_diem(diem, he_so=1):
    return diem * he_so

print(tinh_diem(8))        # 8 (he_so mặc định = 1)
print(tinh_diem(8, 2))     # 16

# === Lambda (hàm ẩn danh) ===
binh_phuong = lambda x: x ** 2
print("Lambda 5²:", binh_phuong(5))

# === *args và **kwargs ===
def tong(*args):
    return sum(args)

print("Tong:", tong(1, 2, 3, 4, 5))

# === Hàm đệ quy ===
def giai_thua(n):
    if n <= 1:
        return 1
    return n * giai_thua(n - 1)

print("5! =", giai_thua(5))
Xin chao, An! 8 16 Lambda 5²: 25 Tong: 15 5! = 120

Lập Trình Hướng Đối Tượng (OOP)

Class, __init__, Method ▼
Python 
class SinhVien:
    """Class Sinh Vien"""

    # Biến class (chung cho mọi đối tượng)
    truong = "NTU"

    def __init__(self, ten, tuoi, diem):
        # Biến instance (riêng cho từng đối tượng)
        self.ten = ten
        self.tuoi = tuoi
        self.__diem = diem  # Private (name mangling)

    def gioi_thieu(self):
        print(f"Toi la {self.ten}, {self.tuoi} tuoi")
        print(f"Hoc tai: {self.truong}")

    @property
    def diem(self):
        return self.__diem

    def __str__(self):
        return f"SinhVien({self.ten}, {self.tuoi}, {self.__diem})"

# Sử dụng
sv = SinhVien("Nguyen Van A", 20, 8.5)
sv.gioi_thieu()
print("Diem:", sv.diem)
print(sv)
Toi la Nguyen Van A, 20 tuoi Hoc tai: NTU Diem: 8.5 SinhVien(Nguyen Van A, 20, 8.5)
Kế thừa & Đa hình ▼
Python 
from abc import ABC, abstractmethod

# Lớp trừu tượng (Abstract Class)
class Shape(ABC):
    @abstractmethod
    def area(self):
        pass

    @abstractmethod
    def perimeter(self):
        pass

    def describe(self):
        print(f"Dien tich: {self.area():.2f}")
        print(f"Chu vi: {self.perimeter():.2f}")

# Lớp con
class HinhTron(Shape):
    def __init__(self, r):
        self.r = r

    def area(self):
        return 3.14159 * self.r ** 2

    def perimeter(self):
        return 2 * 3.14159 * self.r

class HinhChuNhat(Shape):
    def __init__(self, dai, rong):
        self.dai = dai
        self.rong = rong

    def area(self):
        return self.dai * self.rong

    def perimeter(self):
        return 2 * (self.dai + self.rong)

# Đa hình
shapes = [HinhTron(5), HinhChuNhat(4, 6)]
for s in shapes:
    print(f"\n--- {s.__class__.__name__} ---")
    s.describe()
--- HinhTron --- Dien tich: 78.54 Chu vi: 31.42 --- HinhChuNhat --- Dien tich: 24.00 Chu vi: 20.00

Đọc / Ghi File

Python 
# === Ghi file ===
with open("data.txt", "w", encoding="utf-8") as f:
    f.write("Dong 1: Hello Python\n")
    f.write("Dong 2: File I/O\n")
    f.write("Dong 3: Xin chao\n")

print("Da ghi file thanh cong!")

# === Đọc file ===
with open("data.txt", "r", encoding="utf-8") as f:
    # Đọc từng dòng
    for line in f:
        print(line.strip())

# === Đọc tất cả vào list ===
with open("data.txt", "r") as f:
    lines = f.readlines()
    print(f"\nTong so dong: {len(lines)}")
Da ghi file thanh cong! Dong 1: Hello Python Dong 2: File I/O Dong 3: Xin chao Tong so dong: 3
