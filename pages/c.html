C Ngôn Ngữ Lập Trình C

Ngôn ngữ lập trình nền tảng, gần phần cứng - được phát triển bởi Dennis Ritchie tại Bell Labs năm 1972.

Giới Thiệu Ngôn Ngữ C

C là ngôn ngữ lập trình cấp thấp (low-level), mạnh mẽ và hiệu quả. C là nền tảng cho nhiều ngôn ngữ hiện đại như C++, Java, và Python.

📌 Tại sao học C?
  • Hiểu cách máy tính hoạt động ở mức thấp
  • Nền tảng cho C++, Java, và nhiều ngôn ngữ khác
  • Dùng trong lập trình hệ thống, nhúng, game engine
  • Hiệu suất cao, kiểm soát bộ nhớ trực tiếp
C - Hello World 
#include <stdio.h>  // Thư viện nhập/xuất chuẩn

int main() {           // Hàm chính - điểm bắt đầu chương trình
    printf("Hello, World!\n");  // In ra màn hình
    return 0;            // Trả về 0 = chương trình chạy thành công
}
Hello, World!

Cú Pháp Cơ Bản

Cấu trúc chương trình C ▼

Mỗi chương trình C có cấu trúc cơ bản: khai báo thư viện, hàm main, các câu lệnh.

C 
#include <stdio.h>   // 1. Khai báo thư viện

// 2. Khai báo hằng số
#define PI 3.14159

// 3. Khai báo hàm phụ (nếu có)
void chaoMung() {
    printf("Chao mung ban den voi C!\n");
}

// 4. Hàm main - điểm bắt đầu
int main() {
    chaoMung();
    printf("PI = %.5f\n", PI);
    return 0;
}
Chao mung ban den voi C! PI = 3.14159
Nhập / Xuất dữ liệu (scanf / printf) ▼
C 
#include <stdio.h>

int main() {
    char ten[50];
    int tuoi;

    printf("Nhap ten: ");
    scanf("%s", ten);          // Đọc chuỗi

    printf("Nhap tuoi: ");
    scanf("%d", &tuoi);         // Đọc số nguyên

    printf("Xin chao %s, ban %d tuoi!\n", ten, tuoi);
    return 0;
}
Nhap ten: An Nhap tuoi: 20 Xin chao An, ban 20 tuoi!

Biến & Kiểu Dữ Liệu

C là ngôn ngữ kiểu tĩnh - bạn phải khai báo kiểu dữ liệu trước khi sử dụng biến.

C 
#include <stdio.h>

int main() {
    // Kiểu số nguyên
    int soNguyen = 42;
    short soNgan = 100;
    long soDai = 1000000;

    // Kiểu số thực
    float soThuc = 3.14;
    double soThucLon = 3.14159265358979;

    // Kiểu ký tự
    char kyTu = 'A';
    char chuoi[] = "Hello";

    printf("int: %d\n", soNguyen);
    printf("float: %.2f\n", soThuc);
    printf("double: %.14f\n", soThucLon);
    printf("char: %c\n", kyTu);
    printf("string: %s\n", chuoi);
    printf("sizeof(int): %lu bytes\n", sizeof(int));
    printf("sizeof(double): %lu bytes\n", sizeof(double));

    return 0;
}
int: 42 float: 3.14 double: 3.14159265358979 char: A string: Hello sizeof(int): 4 bytes sizeof(double): 8 bytes

Vòng lặp for, while, do...while ▼
C 
#include <stdio.h>

int main() {
    // Vòng lặp for: in bảng cửu chương 5
    printf("--- Bang cuu chuong 5 ---\n");
    for (int i = 1; i <= 5; i++) {
        printf("5 x %d = %d\n", i, 5 * i);
    }

    // Vòng lặp while: đếm ngược
    printf("\n--- Dem nguoc ---\n");
    int n = 5;
    while (n > 0) {
        printf("%d ", n);
        n--;
    }
    printf("\nPhong!\n");

    return 0;
}
--- Bang cuu chuong 5 --- 5 x 1 = 5 5 x 2 = 10 5 x 3 = 15 5 x 4 = 20 5 x 5 = 25 --- Dem nguoc --- 5 4 3 2 1 Phong!

Hàm (Functions)

Hàm giúp chia chương trình thành các phần nhỏ, dễ quản lý và tái sử dụng.

C 
#include <stdio.h>

// Hàm tính diện tích hình chữ nhật
float dienTich(float dai, float rong) {
    return dai * rong;
}

// Hàm tính chu vi
float chuVi(float dai, float rong) {
    return 2 * (dai + rong);
}

// Hàm hoán đổi giá trị (dùng con trỏ)
void hoanDoi(int *a, int *b) {
    int temp = *a;
    *a = *b;
    *b = temp;
}

int main() {
    printf("Dien tich: %.1f\n", dienTich(5, 3));
    printf("Chu vi: %.1f\n", chuVi(5, 3));

    int x = 10, y = 20;
    printf("Truoc: x=%d, y=%d\n", x, y);
    hoanDoi(&x, &y);
    printf("Sau:   x=%d, y=%d\n", x, y);

    return 0;
}
Dien tich: 15.0 Chu vi: 16.0 Truoc: x=10, y=20 Sau: x=20, y=10

Mảng (Arrays)

C 
#include <stdio.h>

int main() {
    // Khai báo và khởi tạo mảng
    int diem[] = {85, 92, 76, 88, 95};
    int n = sizeof(diem) / sizeof(diem[0]);

    // Tìm max, min, tổng
    int max = diem[0], min = diem[0], tong = 0;

    for (int i = 0; i < n; i++) {
        printf("diem[%d] = %d\n", i, diem[i]);
        tong += diem[i];
        if (diem[i] > max) max = diem[i];
        if (diem[i] < min) min = diem[i];
    }

    printf("Max: %d\n", max);
    printf("Min: %d\n", min);
    printf("Trung binh: %.1f\n", (float)tong / n);

    return 0;
}
diem[0] = 85 diem[1] = 92 diem[2] = 76 diem[3] = 88 diem[4] = 95 Max: 95 Min: 76 Trung binh: 87.2

Con Trỏ (Pointers)

Con trỏ lưu trữ địa chỉ bộ nhớ. Đây là tính năng đặc trưng và mạnh mẽ nhất của C.

⚠️ Cẩn thận với con trỏ: Sử dụng sai con trỏ có thể gây ra segmentation fault, memory leak, hoặc undefined behavior!
C 
#include <stdio.h>
#include <stdlib.h>

int main() {
    // === Con trỏ cơ bản ===
    int x = 42;
    int *ptr = &x;    // ptr trỏ đến x

    printf("x = %d\n", x);
    printf("&x = %p\n", &x);
    printf("ptr = %p\n", ptr);
    printf("*ptr = %d\n", *ptr);

    // === Cấp phát bộ nhớ động ===
    int *arr = (int *)malloc(3 * sizeof(int));

    arr[0] = 10;
    arr[1] = 20;
    arr[2] = 30;

    for (int i = 0; i < 3; i++) {
        printf("arr[%d] = %d\n", i, arr[i]);
    }

    free(arr);  // Giải phóng bộ nhớ
    return 0;
}
x = 42 &x = 0x7ffd5e8c3abc ptr = 0x7ffd5e8c3abc *ptr = 42 arr[0] = 10 arr[1] = 20 arr[2] = 30
