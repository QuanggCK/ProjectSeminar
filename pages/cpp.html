C++ Ngôn Ngữ Lập Trình C++

Mở rộng của C với OOP, templates và STL - được phát triển bởi Bjarne Stroustrup từ năm 1979.

Giới Thiệu C++

C++ là ngôn ngữ lập trình đa mô hình (multi-paradigm), hỗ trợ lập trình thủ tục, hướng đối tượng và generic. C++ được sử dụng rộng rãi trong game, hệ điều hành, và ứng dụng hiệu suất cao.

📌 C++ so với C:
  • C++ hỗ trợ OOP (Class, Object, Kế thừa...)
  • C++ có Templates cho lập trình generic
  • C++ có STL (Standard Template Library)
  • C++ dùng cout/cin thay cho printf/scanf
C++ - Hello World 
#include <iostream>    // Thư viện nhập/xuất
using namespace std;    // Sử dụng namespace chuẩn

int main() {
    cout << "Hello, C++!" << endl;
    return 0;
}
Hello, C++!

Cú Pháp Cơ Bản C++

Nhập / Xuất với cin / cout ▼
C++ 
#include <iostream>
#include <string>
using namespace std;

int main() {
    string ten;
    int tuoi;

    cout << "Nhap ten: ";
    getline(cin, ten);        // Đọc cả dòng (có khoảng trắng)

    cout << "Nhap tuoi: ";
    cin >> tuoi;              // Đọc số nguyên

    cout << "Xin chao " << ten
         << ", ban " << tuoi << " tuoi!" << endl;

    return 0;
}
Nhap ten: Nguyen Van A Nhap tuoi: 20 Xin chao Nguyen Van A, ban 20 tuoi!
Tham chiếu (References) & Tham trị ▼
C++ 
#include <iostream>
using namespace std;

// Truyền tham trị - không thay đổi biến gốc
void tangGiaTri(int x) {
    x += 10;
}

// Truyền tham chiếu - thay đổi biến gốc
void tangGiaTriRef(int &x) {
    x += 10;
}

int main() {
    int a = 5;

    tangGiaTri(a);
    cout << "Sau tham tri: " << a << endl;   // 5 (không đổi)

    tangGiaTriRef(a);
    cout << "Sau tham chieu: " << a << endl; // 15 (đã đổi)

    return 0;
}
Sau tham tri: 5 Sau tham chieu: 15

Lập Trình Hướng Đối Tượng (OOP)

OOP là paradigm lập trình chính của C++. Mọi thứ được tổ chức thành Class (lớp) và Object (đối tượng).

Class & Object cơ bản ▼
C++ 
#include <iostream>
#include <string>
using namespace std;

class SinhVien {
private:                        // Đóng gói (Encapsulation)
    string maSV;
    float diemTB;

public:
    string ten;

    // Constructor
    SinhVien(string t, string ma, float diem) {
        ten = t;
        maSV = ma;
        diemTB = diem;
    }

    // Phương thức
    void hienThi() {
        cout << "Ten: " << ten << endl;
        cout << "Ma SV: " << maSV << endl;
        cout << "Diem TB: " << diemTB << endl;
    }

    // Getter
    float getDiemTB() { return diemTB; }
};

int main() {
    SinhVien sv1("Nguyen Van A", "SV001", 8.5);
    sv1.hienThi();
    return 0;
}
Ten: Nguyen Van A Ma SV: SV001 Diem TB: 8.5
Kế thừa (Inheritance) & Đa hình (Polymorphism) ▼
C++ 
#include <iostream>
using namespace std;

// Lớp cha
class HinhHoc {
public:
    virtual float dienTich() = 0;  // Hàm thuần ảo (abstract)
    virtual void hienThi() {
        cout << "Dien tich: " << dienTich() << endl;
    }
};

// Lớp con: Hình tròn
class HinhTron : public HinhHoc {
    float banKinh;
public:
    HinhTron(float r) : banKinh(r) {}
    float dienTich() override {
        return 3.14159 * banKinh * banKinh;
    }
};

// Lớp con: Hình chữ nhật
class HinhChuNhat : public HinhHoc {
    float dai, rong;
public:
    HinhChuNhat(float d, float r) : dai(d), rong(r) {}
    float dienTich() override {
        return dai * rong;
    }
};

int main() {
    HinhTron ht(5);
    HinhChuNhat hcn(4, 6);

    cout << "=== Hinh tron ===" << endl;
    ht.hienThi();

    cout << "=== Hinh chu nhat ===" << endl;
    hcn.hienThi();

    return 0;
}
=== Hinh tron === Dien tich: 78.5398 === Hinh chu nhat === Dien tich: 24

Templates (Khuôn mẫu)

Templates cho phép viết code tổng quát, hoạt động với nhiều kiểu dữ liệu khác nhau.

C++ 
#include <iostream>
using namespace std;

// Function template - Hàm tìm max
template <typename T>
T timMax(T a, T b) {
    return (a > b) ? a : b;
}

// Class template - Cặp giá trị
template <typename T1, typename T2>
class Cap {
public:
    T1 first;
    T2 second;
    Cap(T1 a, T2 b) : first(a), second(b) {}
    void hienThi() {
        cout << "(" << first << ", " << second << ")" << endl;
    }
};

int main() {
    cout << "Max(3, 7): " << timMax(3, 7) << endl;
    cout << "Max(3.14, 2.71): " << timMax(3.14, 2.71) << endl;

    Cap<string, int> sv("Nguyen Van A", 20);
    sv.hienThi();

    return 0;
}
Max(3, 7): 7 Max(3.14, 2.71): 3.14 (Nguyen Van A, 20)

STL (Standard Template Library)

STL cung cấp các cấu trúc dữ liệu và thuật toán sẵn có: vector, map, set, sort, find...

C++ 
#include <iostream>
#include <vector>
#include <algorithm>
#include <map>
using namespace std;

int main() {
    // === Vector ===
    vector<int> ds = {5, 2, 8, 1, 9};

    sort(ds.begin(), ds.end());  // Sắp xếp

    cout << "Vector sau sap xep: ";
    for (int x : ds) cout << x << " ";
    cout << endl;

    // === Map ===
    map<string, int> diemSV;
    diemSV["An"] = 85;
    diemSV["Binh"] = 92;
    diemSV["Cuong"] = 78;

    cout << "\nDiem sinh vien:" << endl;
    for (auto &[ten, diem] : diemSV) {
        cout << ten << ": " << diem << endl;
    }

    return 0;
}
Vector sau sap xep: 1 2 5 8 9 Diem sinh vien: An: 85 Binh: 92 Cuong: 78
